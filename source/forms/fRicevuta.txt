Sub SommaRicevuta()
Dim sqlc As String
Dim rsR As Recordset, db As Database

Set db = CurrentDb


sqlc = "SELECT Ricevuta.IdP, Sum(Ricevuta.Pagato) AS SommaDiPagato, " & _
                "Sum(Causale.Importo) AS SommaDiImporto " & _
        "FROM Ricevuta INNER JOIN Causale ON Ricevuta.IdC = Causale.Id " & _
        "GROUP BY Ricevuta.IdP " & _
            "HAVING Ricevuta.IdP=" & IdP
            
Set rsR = db.OpenRecordset(sqlc, dbOpenDynaset)

If Not rsR.EOF Then
    txtRicevuta = rsR("SommaDiImporto")
    
End If




End Sub
